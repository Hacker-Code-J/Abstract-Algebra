\defbox[Category]{\begin{definition}
A \textbf{category} $\mathcal{C}$ consists of the following data: \begin{enumerate}[(i)]
\item A class $\text{Ob}(\mathcal{C})$, called \textit{objects} of category $\mathcal{C}$.
\item For a pair $A$ and $B$ of objects, a class $\text{Mor}_\mathcal{C}(A,B)$\footnote{Sometimes, we also denote it by $\text{Hom}(A,B)$ or $\mathcal{C}(A,B)$.} whose elements are called \textit{morphisms} (or \textit{arrows}) from $A$ to $B$.
\item For a pair of morphisms $f$ in $\text{Mor}_\mathcal{C}(A,B)$ and $g$ in $\text{Mor}_\mathcal{C}(B,C)$, there is a \textit{composition} $ g\circ f$ in $\text{Mor}_\mathcal{C}(A,C)$.
\end{enumerate}
These data are to satisfy the following rules:
\begin{enumerate}[(i)]
\item For every object $A$, there is an identity morphisms $1_A$ in $\text{Mor}_\mathcal{C}(A,A)$.
\item Composition is associative.
\end{enumerate}
\end{definition}}

\defbox[Functor]{\begin{definition}
A \textbf{functor} $\textbf{F}:\mathcal{C}\to\mathcal{D}$ between two categories $\mathcal{C}$ and $\mathcal{D}$ is given by the following data: \begin{enumerate}[(i)]
	\item A map $\textbf{F}:\text{Ob}(\mathcal{C})\to\text{Ob}(\mathcal{D})$
	\item For every $A$ and $B$ in $\text{Ob}(\mathcal{C})$, a map $\textbf{F}$ given by \[
	\fullfunction{\textbf{F}}{\text{Mor}_{\mathcal{C}}(A,B)}{\text{Mor}_{\mathcal{D}}(\textbf{F}(A), \textbf{F}(B))}{\phi}{\textbf{F}(\phi)}
	\]
\end{enumerate}
\end{definition}}

\defbox[Natural Transformation]{\begin{definition}
Let $\textbf{F},\textbf{G}:\mathcal{C}\to\mathcal{D}$ be functors. A \textbf{natural transformation} (or a \textbf{morphism of functors}) $\eta:\textbf{F}\to\textbf{G}$, is a collection $\set{\eta_A}_{A\in\text{Ob}(\mathcal{C})}$ such that \begin{enumerate}[(i)]
	\item $\eta_A:\textbf{F}(A)\to\textbf{G}(A)$ is a morphism in the category $\mathcal{D}$, and
	\item for each morphism $\phi:A\to B$ of $\mathcal{C}$ the following diagram is commutative
	\begin{center}
		\includegraphics[scale=1.5]{./nt.pdf}
	\end{center}
\end{enumerate}

\end{definition}}
